% Icona circolare: cerchio e glifo sono due nodi distinti.
%
% Centratura del glifo. I glifi hanno altezza e profondita' diverse
% fra loro, quindi centrare la loro scatola non basta: va centrato
% il *disegno*. \ecvglyphcenter misura altezza e profondita' del
% glifo gia' composto e lo abbassa di (altezza - profondita')/2,
% portando il suo centro ottico sulla linea di base. Il nodo viene
% poi ancorato con "base" al centro del cerchio, quindi il centro
% ottico coincide col centro del cerchio per qualunque icona, a
% qualunque corpo.
%
% Regolazioni rapide:
%   \ecvIconSize   diametro del cerchio
%   \ecvIconGlyph  corpo del glifo (tenerlo intorno al 50% del
%                  diametro: sopra il 60% il segno tocca il bordo)
%   \ecvIconRise   quanto il cerchio sale sopra la linea di base
\newcommand{\ecvIconSize}{13pt}
\newcommand{\ecvIconGlyph}{6.4}
\newcommand{\ecvIconRise}{-0.72ex}

\newcommand{\ecvglyphcenter}[1]{\raisebox{\dimexpr(\depth-\height)/2\relax}{#1}}

\newcommand{\ecvicon}[1]{%
  \begin{tikzpicture}[baseline=\ecvIconRise]
    \node[circle,fill=ecvblue,minimum size=\ecvIconSize,inner sep=0pt] (ecvic) {};
    \node[anchor=base,inner sep=0pt,outer sep=0pt,text=white,
          font=\fontsize{\ecvIconGlyph}{\ecvIconGlyph}\selectfont]
      at (ecvic.center) {\ecvnoextract{\ecvglyphcenter{#1}}};
  \end{tikzpicture}%
}

% Sagoma viola con foto ritagliata in cerchio.
%
%   \ecvphoto{foto.jpg}        inquadratura piena (nessun ingrandimento)
%   \ecvphoto[1.25]{foto.jpg}  ingrandisce la foto del 25%
%
% Lo spessore dell'anello colorato e' la differenza fra il raggio
% della sagoma (\ecvBlobScale) e \ecvPhotoR. Alzare \ecvPhotoR o
% abbassare \ecvBlobScale assottiglia l'anello: sotto i ~0,12 cm
% di margine minimo la foto rischia di sbordare dalla sagoma.
% Metti \ecvBlobDebugtrue per sovrapporre alla sagoma il cerchio
% del ritaglio e i dodici punti di controllo.
\newif\ifecvBlobDebug \ecvBlobDebugfalse

\newcommand{\ecvBlobScale}{1.00}  % dimensione della sagoma colorata
\newcommand{\ecvPhotoR}{1.42}     % raggio del ritaglio della foto, in cm
\newcommand{\ecvPhotoDx}{0.05}    % spostamento orizzontale, in cm
\newcommand{\ecvPhotoDy}{0.16}    % spostamento verticale, in cm

% Il profilo e' definito da dodici raggi misurati DAL CENTRO DELLA
% FOTO, non dall'origine della figura. Il \makebox[\linewidth][c]
% centra il disegno nella colonna: senza, la scatola si appoggia al
% margine sinistro e lascia tutto lo spazio residuo a destra.
\NewDocumentCommand{\ecvphoto}{O{1.0} m}{%
  \pgfmathsetlengthmacro{\ecvPhotoW}{#1*2*\ecvPhotoR cm}%
  \par\noindent\makebox[\linewidth][c]{%
    \begin{tikzpicture}
      \fill[ecvnavy,shift={(\ecvPhotoDx,\ecvPhotoDy)},scale=\ecvBlobScale]
      plot[smooth cycle,tension=0.78] coordinates
      {(1.600,0.000) (1.524,0.880) (0.970,1.680) (0.000,1.820)
       (-0.830,1.438) (-1.386,0.800) (-1.620,0.000) (-1.377,-0.795)
       (-0.790,-1.368) (0.000,-1.630) (0.870,-1.507) (1.412,-0.815)};
      \IfFileExists{#2}{%
        \begin{scope}
          \clip (\ecvPhotoDx,\ecvPhotoDy) circle (\ecvPhotoR);
          \node at (\ecvPhotoDx,\ecvPhotoDy) {\includegraphics[width=\ecvPhotoW]{#2}};
        \end{scope}}{}%
      \ifecvBlobDebug
        \begin{scope}[shift={(\ecvPhotoDx,\ecvPhotoDy)}]
          \draw[red,thick] (0,0) circle (\ecvPhotoR);
          \foreach \rr in {1.6,1.8,2.0}{\draw[gray!50,dashed] (0,0) circle (\rr);}
          \foreach \a/\rr in {0/1.60,30/1.76,60/1.94,90/1.82,120/1.66,150/1.60,
                              180/1.62,210/1.59,240/1.58,270/1.63,300/1.74,330/1.63}{%
            \fill[red] (\a:\rr) circle (0.04);
            \node[red,font=\tiny,inner sep=1pt] at (\a:\rr+0.22) {\a};}
        \end{scope}
      \fi
    \end{tikzpicture}}\par
}
