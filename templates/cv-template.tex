  \item\small{
    {#1 \vspace{-3pt}}
  }
}

% v3.3 — same auto-needspace guard as the project headings (v3.1), so an
% Experience/Education subheading never lands alone at the foot of a page
% with its bullets orphaned onto the next one.
\newcommand{\resumeSubheading}[4]{
  \needspace{6\baselineskip}%
  \vspace{-3pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{\color{TextDark}#1} & {\color{TextMuted}\textbf{\small #2}} \\
      \textit{\small #3} & {\color{TextMuted}\textit{\small #4}} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubheadingContinue}[2]{
  \vspace{-3pt}
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textit{\small #1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

% v3.1 — auto-needspace prefix to reduce mid-project page splits.
% Reserves ~6 baseline-heights of vertical space; if not available, page break.
\newcommand{\resumeProjectHeading}[2]{
  \needspace{6\baselineskip}%
  \vspace{-3pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{\color{TextDark}#1} & \textbf{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

% v3.2 — project heading with GitHub icon + clickable project title.
% Use this for any project that has a public repo URL.
%   #1 = project name (LaTeX-escaped)   #2 = repo URL (raw, NOT escaped)
%   #3 = context, e.g. "$|$ Automated Parasite Detection" (escaped)
%   #4 = right-aligned badge/date (escaped)
\newcommand{\resumeProjectHeadingLink}[4]{
  \needspace{6\baselineskip}%
  \vspace{-3pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{{\color{BrandAccent}\faGithub}\ \href{#2}{\color{TextDark}#1}\ \emph{\small\color{TextMuted}#3}} & \textbf{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}
